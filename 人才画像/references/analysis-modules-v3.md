### 2.2 薪资分位计算

```python
# 薪资分位计算公式
def calculate_salary_percentile(salaries: List[float], percentile: int) -> float:
    """
    计算薪资分位值
    :param salaries: 薪资列表
    :param percentile: 分位数（25/50/75/90）
    :return: 分位薪资值
    """
    sorted_salaries = sorted(salaries)
    index = int(len(sorted_salaries) * percentile / 100)
    return sorted_salaries[min(index, len(sorted_salaries) - 1)]

# 分位值说明
```

### 3.2 供需比计算

```python
# 供需比计算公式
def calculate_supply_demand_ratio(supply: int, demand: int) -> Dict:
    """
    计算供需比
    :param supply: 候选人数量（供给）
    :param demand: 职位数量（需求）
    :return: 供需分析结果
    """
    ratio = supply / demand if demand > 0 else 0
    
    if ratio >= 2.0:
        status = "供给过剩"
        urgency = "低"
        recommendation = "控制招聘规模，提高筛选标准"
    elif ratio >= 1.0:
        status = "基本平衡"
        urgency = "中"
        recommendation = "正常招聘节奏"
    elif ratio >= 0.5:
        status = "轻度紧缺"
        urgency = "高"
        recommendation = "加快招聘，优化条件"
    else:
        status = "严重紧缺"
        urgency = "紧急"
        recommendation = "战略储备，开辟新渠道"
    
    return {
        "供需比": round(ratio, 2),
        "状态": status,
        "紧急程度": urgency,
        "建议": recommendation
    }
```

## 4. 分析模块集成

### 4.1 模块调用接口

```python
# 分析模块集成接口
class AnalysisModuleIntegration:
    
    def run_talent_flow_analysis(self, data: Dict, config: Dict) -> Dict:
        """执行人才流动分析"""
        # 1. 数据准备
        flow_data = self.prepare_flow_data(data)
        
        # 2. 指标计算
        indicators = self.calculate_flow_indicators(flow_data)
        
        # 3. 趋势分析
        trends = self.analyze_flow_trends(flow_data)
        
        # 4. 洞察生成
        insights = self.generate_flow_insights(indicators, trends)
        
        return {
            "indicators": indicators,
            "trends": trends,
            "insights": insights
        }
    
    def run_salary_benchmark_analysis(self, data: Dict, config: Dict) -> Dict:
        """执行薪资benchmark分析"""
        # 1. 数据标准化
        salary_data = self.standardize_salary_data(data)
        
        # 2. 分位计算
        percentiles = self.calculate_percentiles(salary_data)
        
        # 3. 预测建模
        predictions = self.predict_salary_trend(salary_data, config)
        
        # 4. 报告生成
        return {
            "percentiles": percentiles,
            "predictions": predictions,
            "recommendations": self.generate_recommendations(percentiles, predictions)
        }
    
    def run_supply_demand_analysis(self, data: Dict, config: Dict) -> Dict:
        """执行供需分析"""
        # 1. 供给分析
        supply = self.analyze_supply(data)
        
        # 2. 需求分析
        demand = self.analyze_demand(data)
        
        # 3. 供需比计算
        ratio = self.calculate_supply_demand_ratio(supply, demand)
        
        # 4. 趋势预测
        predictions = self.predict_supply_demand_trend(supply, demand)
        
        return {
            "supply": supply,
            "demand": demand,
            "ratio": ratio,
            "predictions": predictions
        }
```

## 6. 常见问题与解决方案

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| 薪资数据缺失多 | 历史数据未录入 | 插值估算+标注 |
